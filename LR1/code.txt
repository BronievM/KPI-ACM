import tkinter as tk
from tkinter import ttk, filedialog, messagebox
import math
import json

class CalculatorLogic:
    @staticmethod
    def calculate_linear(a, b, c):
        if a < 0:
            raise ValueError("Параметр 'a' повинен бути >= 0")
        
        sqrt_a = math.sqrt(a)
        term_b = b + 5 * sqrt_a
        
        if term_b < 0:
            raise ValueError("Вираз (b + 5*sqrt(a)) повинен бути >= 0")
            
        sqrt_term_b = math.sqrt(term_b)
        base = 5 + c * sqrt_term_b
        
        if base >= 0:
            return math.pow(base, 1/3)
        else:
            return -math.pow(abs(base), 1/3)

    @staticmethod
    def calculate_branching(k, d):
        if k > 10:
            val = k * abs(d) + d * abs(k)
            if val < 0:
                raise ValueError("Підкореневий вираз від'ємний")
            return math.sqrt(val)
        else:
            return (k + d) ** 2

    @staticmethod
    def calculate_cyclic(n, a_list, b_list):
        if len(a_list) < n + 1 or len(b_list) < n + 1:
            raise ValueError(f"Довжина масивів має бути мінімум {n+1}")
            
        f = 1.0
        for i in range(1, n + 1):
            idx_i = i - 1
            term1 = a_list[idx_i] + b_list[idx_i + 1]
            
            inner_sum = 0.0
            for j in range(i, n + 1):
                idx_j = j - 1
                inner_sum += a_list[idx_j + 1] * b_list[idx_j]
            
            f *= (term1 + inner_sum)
            
        return f

class MainWindow(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("Лабораторна робота - Варіант 9")
        self.geometry("600x450")
        
        self.notebook = ttk.Notebook(self)
        self.notebook.pack(expand=True, fill='both', padx=10, pady=10)
        
        self.init_linear_tab()
        self.init_branching_tab()
        self.init_cyclic_tab()

    def init_linear_tab(self):
        frame = ttk.Frame(self.notebook)
        self.notebook.add(frame, text="Лінійний")
        
        self.ln_entries = {}
        for var in ['a', 'b', 'c']:
            row = ttk.Frame(frame)
            row.pack(fill='x', padx=20, pady=5)
            ttk.Label(row, text=f"{var}:").pack(side=tk.LEFT)
            entry = ttk.Entry(row)
            entry.pack(side=tk.RIGHT, expand=True, fill='x')
            self.ln_entries[var] = entry

        btn_row = ttk.Frame(frame)
        btn_row.pack(pady=10)
        ttk.Button(btn_row, text="Обчислити", command=self.run_linear).pack(side=tk.LEFT, padx=5)
        ttk.Button(btn_row, text="Завантажити JSON", command=lambda: self.load_data('linear')).pack(side=tk.LEFT, padx=5)
        
        self.ln_res = ttk.Label(frame, text="Результат: ")
        self.ln_res.pack(pady=10)

    def init_branching_tab(self):
        frame = ttk.Frame(self.notebook)
        self.notebook.add(frame, text="Розгалужений")
        
        self.br_entries = {}
        for var in ['k', 'd']:
            row = ttk.Frame(frame)
            row.pack(fill='x', padx=20, pady=5)
            ttk.Label(row, text=f"{var}:").pack(side=tk.LEFT)
            entry = ttk.Entry(row)
            entry.pack(side=tk.RIGHT, expand=True, fill='x')
            self.br_entries[var] = entry

        btn_row = ttk.Frame(frame)
        btn_row.pack(pady=10)
        ttk.Button(btn_row, text="Обчислити", command=self.run_branching).pack(side=tk.LEFT, padx=5)
        ttk.Button(btn_row, text="Завантажити JSON", command=lambda: self.load_data('branching')).pack(side=tk.LEFT, padx=5)
        
        self.br_res = ttk.Label(frame, text="Результат: ")
        self.br_res.pack(pady=10)

    def init_cyclic_tab(self):
        frame = ttk.Frame(self.notebook)
        self.notebook.add(frame, text="Циклічний")
        
        self.cy_entries = {}
        
        row_n = ttk.Frame(frame)
        row_n.pack(fill='x', padx=20, pady=5)
        ttk.Label(row_n, text="n:").pack(side=tk.LEFT)
        entry_n = ttk.Entry(row_n)
        entry_n.pack(side=tk.RIGHT, expand=True, fill='x')
        self.cy_entries['n'] = entry_n
        
        row_a = ttk.Frame(frame)
        row_a.pack(fill='x', padx=20, pady=5)
        ttk.Label(row_a, text="Масив a (через кому):").pack(side=tk.LEFT)
        entry_a = ttk.Entry(row_a)
        entry_a.pack(side=tk.RIGHT, expand=True, fill='x')
        self.cy_entries['a'] = entry_a
        
        row_b = ttk.Frame(frame)
        row_b.pack(fill='x', padx=20, pady=5)
        ttk.Label(row_b, text="Масив b (через кому):").pack(side=tk.LEFT)
        entry_b = ttk.Entry(row_b)
        entry_b.pack(side=tk.RIGHT, expand=True, fill='x')
        self.cy_entries['b'] = entry_b

        btn_row = ttk.Frame(frame)
        btn_row.pack(pady=10)
        ttk.Button(btn_row, text="Обчислити", command=self.run_cyclic).pack(side=tk.LEFT, padx=5)
        ttk.Button(btn_row, text="Завантажити JSON", command=lambda: self.load_data('cyclic')).pack(side=tk.LEFT, padx=5)
        
        self.cy_res = ttk.Label(frame, text="Результат: ")
        self.cy_res.pack(pady=10)

    def run_linear(self):
        try:
            a = float(self.ln_entries['a'].get())
            b = float(self.ln_entries['b'].get())
            c = float(self.ln_entries['c'].get())
            res = CalculatorLogic.calculate_linear(a, b, c)
            self.ln_res.config(text=f"Результат: {res:.4f}")
        except ValueError as e:
            messagebox.showerror("Помилка", str(e))

    def run_branching(self):
        try:
            k = float(self.br_entries['k'].get())
            d = float(self.br_entries['d'].get())
            res = CalculatorLogic.calculate_branching(k, d)
            self.br_res.config(text=f"Результат: {res:.4f}")
        except ValueError as e:
            messagebox.showerror("Помилка", str(e))

    def run_cyclic(self):
        try:
            n = int(self.cy_entries['n'].get())
            a_str = self.cy_entries['a'].get().split(',')
            b_str = self.cy_entries['b'].get().split(',')
            a = [float(x) for x in a_str if x.strip()]
            b = [float(x) for x in b_str if x.strip()]
            
            res = CalculatorLogic.calculate_cyclic(n, a, b)
            self.cy_res.config(text=f"Результат: {res:.4f}")
        except ValueError as e:
            messagebox.showerror("Помилка", str(e))

    def load_data(self, tab_type):
        path = filedialog.askopenfilename(filetypes=[("JSON files", "*.json")])
        if not path:
            return
        try:
            with open(path, 'r', encoding='utf-8') as f:
                data = json.load(f)
            
            if tab_type == 'linear':
                d = data.get('linear', {})
                self.ln_entries['a'].delete(0, tk.END)
                self.ln_entries['a'].insert(0, d.get('a', ''))
                self.ln_entries['b'].delete(0, tk.END)
                self.ln_entries['b'].insert(0, d.get('b', ''))
                self.ln_entries['c'].delete(0, tk.END)
                self.ln_entries['c'].insert(0, d.get('c', ''))
                
            elif tab_type == 'branching':
                d = data.get('branching', {})
                self.br_entries['k'].delete(0, tk.END)
                self.br_entries['k'].insert(0, d.get('k', ''))
                self.br_entries['d'].delete(0, tk.END)
                self.br_entries['d'].insert(0, d.get('d', ''))
                
            elif tab_type == 'cyclic':
                d = data.get('cyclic', {})
                self.cy_entries['n'].delete(0, tk.END)
                self.cy_entries['n'].insert(0, d.get('n', ''))
                self.cy_entries['a'].delete(0, tk.END)
                self.cy_entries['a'].insert(0, ",".join(map(str, d.get('a', []))))
                self.cy_entries['b'].delete(0, tk.END)
                self.cy_entries['b'].insert(0, ",".join(map(str, d.get('b', []))))
                
        except Exception as e:
            messagebox.showerror("Помилка файлу", str(e))

if __name__ == "__main__":
    app = MainWindow()
    app.mainloop()
